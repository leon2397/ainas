### 1.3 连接池

**实现方案**：
```python
# database/db.py
import aiosqlite
from contextlib import asynccontextmanager

class DatabasePool:
    def __init__(self, db_path: str, max_connections: int = 5):
        self.db_path = db_path
        self.max_connections = max_connections
        self.pool = asyncio.Queue(maxsize=max_connections)
        
    async def init(self):
        for _ in range(self.max_connections):
            conn = await aiosqlite.connect(self.db_path)
            await self.pool.put(conn)
    
    @asynccontextmanager
    async def acquire(self):
        conn = await self.pool.get()
        try:
            yield conn
        finally:
            await self.pool.put(conn)
```

## 2️⃣ 媒体处理优化

### 2.1 缩略图缓存

**当前问题**：
- 每次请求都重新生成缩略图
- CPU占用高
- 响应慢

**优化方案**：

```python
# services/thumbnail_cache.py
import hashlib
from pathlib import Path
from functools import lru_cache

class ThumbnailCache:
    def __init__(self, cache_dir: str = "./data/thumbnails"):
        self.cache_dir = Path(cache_dir)
        self.cache_dir.mkdir(exist_ok=True)
        
    def get_cache_key(self, file_path: str, size: tuple) -> str:
        """生成缓存键"""
        content = f"{file_path}:{size[0]}x{size[1]}"
        return hashlib.md5(content.encode()).hexdigest()
    
    def get_thumbnail_path(self, file_path: str, size: tuple) -> Path:
        """获取缩略图路径"""
        cache_key = self.get_cache_key(file_path, size)
        return self.cache_dir / f"{cache_key}.jpg"
    
    def exists(self, file_path: str, size: tuple) -> bool:
        """检查缓存是否存在"""
        return self.get_thumbnail_path(file_path, size).exists()
    
    async def generate(self, file_path: str, size: tuple) -> Path:
        """生成缩略图（带缓存）"""
        thumb_path = self.get_thumbnail_path(file_path, size)
        
        if thumb_path.exists():
            return thumb_path
        
        # 生成新缩略图
        from PIL import Image
        with Image.open(file_path) as img:
            img.thumbnail(size)
            img.save(thumb_path, "JPEG", quality=85)
        
        return thumb_path

# 全局缓存实例
thumbnail_cache = ThumbnailCache()
```

**预期提升**：缩略图加载 10x

### 2.2 批量处理

**当前问题**：
```python
# 低效 - 单文件循环
for file in files:
    await process_single(file)  # 每次数据库插入
```

**优化方案**：
```python
# 高效 - 批量处理
async def batch_process(files: list, batch_size: int = 100):
    for i in range(0, len(files), batch_size):
        batch = files[i:i + batch_size]
        
        # 批量插入
        async with db.transaction() as conn:
            await conn.executemany(
                "INSERT INTO media (...) VALUES (...)",
                [(f.path, f.hash, f.size) for f in batch]
            )
        
        # 批量生成缩略图（并行）
        await asyncio.gather(*[
            thumbnail_cache.generate(f.path, (300, 300))
            for f in batch
        ])
```

def load_image_safe(file_path: str, max_size: tuple = (4096, 4096)):
    """安全加载图片（限制内存）"""
    with Image.open(file_path) as img:
        # 如果图片太大，先缩小

### 3.1 模型缓存

# core/ai_engine.py
from functools import lru_cache

class AIEngine:
    def __init__(self):
        self._model_cache = {}
    
    @lru_cache(maxsize=1)
    def get_model(self, model_name: str):
        """缓存模型实例"""
        if model_name not in self._model_cache:
            self._model_cache[model_name] = self._load_model(model_name)
        return self._model_cache[model_name]
    
    async def classify_batch(self, image_paths: list, batch_size: int = 8):
        """批量分类"""
        results = []
        for i in range(0, len(image_paths), batch_size):
            batch = image_paths[i:i + batch_size]
            batch_results = await self._classify_batch(batch)
            results.extend(batch_results)
        return results
```

# services/classifier.py
import hashlib
import json
from pathlib import Path

class ClassificationCache:
    def __init__(self, cache_dir: str = "./data/classification_cache"):
        self.cache_dir = Path(cache_dir)
        self.cache_dir.mkdir(exist_ok=True)
    
    def _get_cache_key(self, file_hash: str) -> str:
        return hashlib.md5(file_hash.encode()).hexdigest()
    
    def get(self, file_hash: str) -> dict:
        """获取缓存的分类结果"""
        cache_file = self.cache_dir / f"{self._get_cache_key(file_hash)}.json"
        if cache_file.exists():
            return json.loads(cache_file.read_text())
        return None
    
    def set(self, file_hash: str, result: dict):
        """缓存分类结果"""
        cache_file = self.cache_dir / f"{self._get_cache_key(file_hash)}.json"
        cache_file.write_text(json.dumps(result))
```

## 4️⃣ 系统监控增强

### 4.1 性能指标

**实现方案**：
```python
# services/performance_monitor.py
import time
import statistics
from collections import deque

class PerformanceMonitor:
    def __init__(self, window_size: int = 100):
        self.metrics = {
            'api_latency': deque(maxlen=window_size),
            'db_query_time': deque(maxlen=window_size),
            'thumbnail_gen_time': deque(maxlen=window_size),
        }
    
    def record(self, metric_name: str, value: float):
        """记录指标"""
        if metric_name in self.metrics:
            self.metrics[metric_name].append(value)
    
    def get_stats(self, metric_name: str) -> dict:
        """获取统计信息"""
        values = list(self.metrics[metric_name])
        if not values:
            return {}
        return {
            'count': len(values),
            'avg': statistics.mean(values),
            'min': min(values),
            'max': max(values),
            'p95': sorted(values)[int(len(values) * 0.95)] if len(values) > 20 else max(values)
        }

# 装饰器记录API耗时
from functools import wraps

def monitor_performance(metric_name: str):
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            start = time.time()
            result = await func(*args, **kwargs)
            elapsed = time.time() - start
            performance_monitor.record(metric_name, elapsed)
            return result
        return wrapper
    return decorator
```

### 4.2 慢查询日志

**实现方案**：
```python
# database/db.py
import time
import logging

logger = logging.getLogger(__name__)

class SlowQueryLogger:
    def __init__(self, threshold_ms: float = 100):
        self.threshold_ms = threshold_ms
    
    async def execute(self, conn, query: str, params: tuple = None):
        """执行查询并记录慢查询"""
        start = time.time()
        result = await conn.execute(query, params)
        elapsed_ms = (time.time() - start) * 1000
        
        if elapsed_ms > self.threshold_ms:
            logger.warning(
                f"Slow query ({elapsed_ms:.1f}ms): {query[:100]}..."
            )
        
        return result
```
