```python
# plugins/base.py
from abc import ABC, abstractmethod
from typing import Any, Dict, List

class Plugin(ABC):
    """插件基类"""
    
    name: str = ""
    version: str = "1.0.0"
    description: str = ""
    
    @abstractmethod
    async def initialize(self, config: Dict[str, Any]) -> bool:
        """初始化插件"""
        pass
    
    @abstractmethod
    async def shutdown(self):
        """关闭插件"""
        pass
    
    def get_config_schema(self) -> Dict[str, Any]:
        """返回配置 schema"""
        return {}


class StoragePlugin(Plugin):
    """存储插件接口"""
    
    @abstractmethod
    async def connect(self) -> bool:
        """连接存储"""
        pass
    
    @abstractmethod
    async def list_files(self, path: str) -> List[Dict]:
        """列出文件"""
        pass
    
    @abstractmethod
    async def read_file(self, path: str) -> bytes:
        """读取文件"""
        pass
    
    @abstractmethod
    async def write_file(self, path: str, data: bytes) -> bool:
        """写入文件"""
        pass


class AIEnginePlugin(Plugin):
    """AI引擎插件接口"""
    
    @abstractmethod
    async def load_model(self, model_path: str) -> bool:
        """加载模型"""
        pass
    
    @abstractmethod
    async def classify(self, image_path: str) -> List[Dict]:
        """分类图片"""
        pass
    
    @abstractmethod
    async def extract_features(self, image_path: str) -> List[float]:
        """提取特征向量"""
        pass
```

### 1.2 插件加载器

```python
# plugins/loader.py
import importlib
import pkgutil
from pathlib import Path
from typing import Dict, Type, List

from plugins.base import Plugin

class PluginLoader:
    """插件加载器"""
    
    def __init__(self, plugin_dirs: List[str] = None):
        self.plugin_dirs = plugin_dirs or ["./plugins"]
        self.plugins: Dict[str, Plugin] = {}
        self.plugin_classes: Dict[str, Type[Plugin]] = {}
    
    def discover_plugins(self) -> List[str]:
        """发现可用插件"""
        discovered = []
        
        for plugin_dir in self.plugin_dirs:
            path = Path(plugin_dir)
            if not path.exists():
                continue
            
            for item in path.iterdir():
                if item.is_dir() and (item / "__init__.py").exists():
                    discovered.append(item.name)
                elif item.suffix == ".py" and item.stem != "__init__":
                    discovered.append(item.stem)
        
        return discovered
    
    def load_plugin(self, name: str, config: dict = None) -> Plugin:
        """加载插件"""
        if name in self.plugins:
            return self.plugins[name]
        
        # 动态导入
        module = importlib.import_module(f"plugins.{name}")
        
        # 查找插件类
        plugin_class = None
        for attr_name in dir(module):
            attr = getattr(module, attr_name)
            if (isinstance(attr, type) and 
                issubclass(attr, Plugin) and 
                attr != Plugin):
                plugin_class = attr
                break
        
        if not plugin_class:
            raise ValueError(f"No plugin class found in {name}")
        
        # 实例化
        instance = plugin_class()
        self.plugin_classes[name] = plugin_class
        
        # 初始化
        if config:
            instance.initialize(config)
        
        self.plugins[name] = instance
        return instance
    
    def unload_plugin(self, name: str):
        """卸载插件"""
        if name in self.plugins:
            self.plugins[name].shutdown()
            del self.plugins[name]
    
    def get_plugin(self, name: str) -> Plugin:
        """获取插件实例"""
        return self.plugins.get(name)
    
    def list_plugins(self) -> List[Dict]:
        """列出已加载插件"""
        return [
            {
                "name": p.name,
                "version": p.version,
                "description": p.description
            }
            for p in self.plugins.values()
        ]
```

### 2.2 配置热更新

```python
# core/config_manager.py
import os
import yaml
import asyncio
from pathlib import Path
from typing import Any, Dict, Callable
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

class ConfigManager:
    """配置管理器（支持热更新）"""
    
    def __init__(self, config_dir: str = "./config"):
        self.config_dir = Path(config_dir)
        self.configs: Dict[str, Any] = {}
        self.callbacks: Dict[str, List[Callable]] = {}
        self._observer = None
        self._lock = asyncio.Lock()
    
    async def load_all(self):
        """加载所有配置"""
        for config_file in self.config_dir.glob("*.yaml"):
            name = config_file.stem
            await self._load_config(name)
    
    async def _load_config(self, name: str):
        """加载单个配置"""
        config_file = self.config_dir / f"{name}.yaml"
        if not config_file.exists():
            return
        
        async with self._lock:
            with open(config_file, 'r', encoding='utf-8') as f:
                self.configs[name] = yaml.safe_load(f)
        
        # 触发回调
        await self._notify_change(name)
    
    def get(self, name: str, key: str = None, default: Any = None) -> Any:
        """获取配置值"""
        config = self.configs.get(name, {})
        if key is None:
            return config
        
        # 支持点号路径，如 "server.port"
        keys = key.split('.')
        value = config
        for k in keys:
            if isinstance(value, dict):
                value = value.get(k)
            else:
                return default
        
        return value if value is not None else default
    
    def on_change(self, config_name: str, callback: Callable):
        """注册配置变更回调"""
        if config_name not in self.callbacks:
            self.callbacks[config_name] = []
        self.callbacks[config_name].append(callback)
    
    async def _notify_change(self, config_name: str):
        """通知配置变更"""
        for callback in self.callbacks.get(config_name, []):
            try:
                if asyncio.iscoroutinefunction(callback):
                    await callback(self.configs[config_name])
                else:
                    callback(self.configs[config_name])
            except Exception as e:
                print(f"Config callback error: {e}")
    
    def start_watching(self):
        """启动配置文件监控"""
        class ConfigHandler(FileSystemEventHandler):
            def __init__(self, manager):
                self.manager = manager
            
            def on_modified(self, event):
                if event.src_path.endswith('.yaml'):
                    name = Path(event.src_path).stem
                    asyncio.create_task(self.manager._load_config(name))
        
        self._observer = Observer()
        self._observer.schedule(
            ConfigHandler(self),
            str(self.config_dir),
            recursive=False
        )
        self._observer.start()
    
    def stop_watching(self):
        """停止监控"""
        if self._observer:
            self._observer.stop()
            self._observer.join()

# 全局配置管理器
config_manager = ConfigManager()
```

```python
# database/migrations.py
import aiosqlite
from pathlib import Path

class DatabaseMigration:
    """数据库迁移管理"""
    
    def __init__(self, db_path: str):
        self.db_path = db_path
        self.migrations_dir = Path("./database/migrations")
    
    async def init_migration_table(self):
        """初始化迁移表"""
        async with aiosqlite.connect(self.db_path) as db:
            await db.execute("""
                CREATE TABLE IF NOT EXISTS schema_migrations (
                    version INTEGER PRIMARY KEY,
                    applied_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
                    description TEXT
                )
            """)
            await db.commit()
    
    async def get_current_version(self) -> int:
        """获取当前版本"""
        async with aiosqlite.connect(self.db_path) as db:
            cursor = await db.execute(
                "SELECT MAX(version) FROM schema_migrations"
            )
            row = await cursor.fetchone()
            return row[0] or 0
    
    async def migrate(self, target_version: int = None):
        """执行迁移"""
        await self.init_migration_table()
        current = await self.get_current_version()
        
        # 获取所有迁移文件
        migrations = sorted(self.migrations_dir.glob("*.sql"))
        
        for migration_file in migrations:
            version = int(migration_file.stem.split("_")[0])
            
            if version <= current:
                continue
            
            if target_version and version > target_version:
                break
            
            # 执行迁移
            print(f"Applying migration {version}...")
            sql = migration_file.read_text()
            
            async with aiosqlite.connect(self.db_path) as db:
                await db.executescript(sql)
                await db.execute(
                    "INSERT INTO schema_migrations (version, description) VALUES (?, ?)",
                    (version, migration_file.stem)
                )
                await db.commit()
            
            print(f"Migration {version} applied.")
```

```python
from typing import Dict, Any
import redis.asyncio as redis

class StatelessService:
    """无状态服务基类"""
    
    def __init__(self, redis_url: str = "redis://localhost"):
        self.redis = redis.from_url(redis_url)
    
    async def get_session(self, session_id: str) -> Dict[str, Any]:
        """从 Redis 获取会话"""
        data = await self.redis.get(f"session:{session_id}")
        return json.loads(data) if data else {}
    
    async def set_session(self, session_id: str, data: Dict[str, Any], ttl: int = 3600):
        """保存会话到 Redis"""
        await self.redis.setex(
            f"session:{session_id}",
            ttl,
            json.dumps(data)
        )
    
    async def acquire_lock(self, lock_name: str, ttl: int = 30) -> bool:
        """分布式锁"""
        return await self.redis.set(
            f"lock:{lock_name}",
            "1",
            nx=True,
            ex=ttl
        )
    
    async def release_lock(self, lock_name: str):
        """释放锁"""
        await self.redis.delete(f"lock:{lock_name}")
```

```python
import asyncio
from typing import Callable, Any
import redis.asyncio as redis

class DistributedTaskQueue:
    """分布式任务队列"""
    
    def __init__(self, redis_url: str = "redis://localhost"):
        self.redis = redis.from_url(redis_url)
        self.handlers: Dict[str, Callable] = {}
    
    def register_handler(self, task_type: str, handler: Callable):
        """注册任务处理器"""
        self.handlers[task_type] = handler
    
    async def enqueue(self, task_type: str, payload: Any, priority: int = 0):
        """添加任务到队列"""
        task = {
            "type": task_type,
            "payload": payload,
            "priority": priority,
            "created_at": time.time()
        }
        
        # 使用优先级队列
        await self.redis.zadd(
            "task_queue",
            {json.dumps(task): priority}
        )
    
    async def worker(self):
        """工作进程"""
        while True:
            # 获取优先级最高的任务
            tasks = await self.redis.zpopmax("task_queue", count=1)
            
            if not tasks:
                await asyncio.sleep(1)
                continue
```
