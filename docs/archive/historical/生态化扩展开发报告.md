## 🚀 使用方法

### 1. 云存储同步

```python
from plugins.third_party.cloud_storage import (
    BaiduNetdiskProvider, 
    cloud_sync_manager
)

# 配置百度网盘
config = {
    "access_token": "your_token",
    "refresh_token": "your_refresh_token"
}
provider = BaiduNetdiskProvider(config)

# 注册到管理器
cloud_sync_manager.register_provider(provider)

# 创建同步任务
task = await cloud_sync_manager.sync_to_cloud(
    "baidu_netdisk",
    "/local/path",
    "/remote/path"
)
```

### 2. 插件市场

```python
from plugins.marketplace.marketplace import plugin_marketplace, plugin_manager

# 搜索插件
plugins = await plugin_marketplace.search_plugins(
    query="storage",
    plugin_type=PluginType.STORAGE
)

# 安装插件
await plugin_manager.install_plugin("plugin.zip", "plugin_id")

# 启用插件
await plugin_manager.enable_plugin("plugin_id")
```

### 3. 移动同步

```python
from sync.mobile.sync_service import mobile_sync_service

# 注册设备
device = await mobile_sync_service.register_device({
    "user_id": "user_123",
    "device_name": "iPhone 15",
    "device_type": "ios"
})

# 更新同步设置
await mobile_sync_service.update_sync_settings(device.device_id, {
    "enabled": True,
    "sync_photos": True,
    "sync_videos": True
})
```

### 4. 社区功能

```python
from community.service import community_service

# 创建分享
share = await community_service.create_share("user_123", {
    "media_ids": ["media_1", "media_2"],
    "share_type": "public",
    "title": "我的旅行照片"
})

# 添加评论
comment = await community_service.add_comment(
    share.id, "user_456", "照片真好看！"
)

# 点赞
await community_service.like_share(share.id, "user_456")
```

### 5. 外部API调用

```bash
# 获取媒体列表
curl "http://localhost:8000/api/v1/external/media?api_key=your_key&limit=20"

# 搜索媒体
curl "http://localhost:8000/api/v1/external/search?q=vacation&api_key=your_key"

# 获取统计
curl "http://localhost:8000/api/v1/external/stats?api_key=your_key"
```
