## 🚀 使用方法

### 云存储集成

```python
from plugins.third_party.cloud_storage import cloud_sync_manager, BaiduNetdiskProvider

# 连接百度网盘
config = {"access_token": "your_token"}
provider = BaiduNetdiskProvider(config)
await provider.authenticate()
cloud_sync_manager.register_provider(provider)

# 同步文件
task = await cloud_sync_manager.sync_to_cloud(
    "baidu_netdisk",
    "/local/path",
    "/remote/path"
)
```

### 插件市场

```python
from plugins.marketplace.marketplace import plugin_marketplace, plugin_manager

# 搜索插件
plugins = await plugin_marketplace.search_plugins(
    query="storage",
    plugin_type=PluginType.STORAGE
)

# 安装插件
await plugin_manager.install_plugin("plugin.zip", "plugin_id")

# 启用插件
await plugin_manager.enable_plugin("plugin_id")
```

### 移动端同步

```python
from sync.mobile.sync_service import mobile_sync_service

# 注册设备
device = await mobile_sync_service.register_device({
    "user_id": "user_123",
    "device_name": "iPhone 15",
    "device_type": "ios"
})

# 更新设置
await mobile_sync_service.update_sync_settings(device_id, {
    "enabled": True,
    "sync_photos": True,
    "auto_sync": True
})
```

### 社区功能

```python
from community.service import community_service

# 创建分享
share = await community_service.create_share(user_id, {
    "media_ids": ["media_1", "media_2"],
    "title": "旅行照片",
    "share_type": "public"
})

# 添加评论
comment = await community_service.add_comment(share.id, user_id, "太美了！")

# 点赞
await community_service.like_share(share.id, user_id)
```
