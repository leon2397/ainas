#### doLogin 函数（Line 677-698）
```javascript
async function doLogin(e) {
    e.preventDefault();
    const btn = document.getElementById("login-btn");
    btn.disabled = true;
    btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> 登录涓?..';  // ⚠️ 还有乱码
    
    try {
        const form = new FormData();
        form.append("username", document.getElementById("login-username").value);
        form.append("password", document.getElementById("login-password").value);
        const resp = await fetch(API_BASE + "/api/auth/login", {method: "POST", body: form});
        if (!resp.ok) { throw new Error("登录失败"); }
        const data = await resp.json();
        localStorage.setItem("nas_token", data.access_token);  // ⚠️ 存储键名是 nas_token
        currentUser = {username: data.username, role: data.role};
        hideLoginPage();  // ✅ 调用了跳转函数
        initApp();
    } catch(err) {
        document.getElementById("login-error").textContent = "用户名或密码错误";
        document.getElementById("login-error").classList.remove("d-none");
        btn.disabled = false;
        btn.innerHTML = '<i class="fas fa-sign-in-alt"></i> 登录';  // ✅ 重置按钮
    }
}
```
